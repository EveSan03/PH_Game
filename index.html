  <div id="game-container">

    <div id="ui">

      <div id="score">Monedas FESP: 0 / 5</div>

      <div id="level">Nivel 1: Conceptos y FESP</div>

    <canvas id="gameCanvas" width="800" height="400"></canvas>

    ⬅️ ➡️ Flechas / A-D: Moverse | ⬆️ / Espacio: Saltar <br>

    Recoge las <b>Monedas FESP</b> y elimina a los <b>Goombas Biomédicos</b> o salta los virus.

<script>

const canvas = document.getElementById('gameCanvas');

const ctx = canvas.getContext('2d');


// Estado del juego

let score = 0;

let gameOver = false;

let gameWon = false;

let currentLevel = 1;

let message = "";


// Teclas

const keys = { right: false, left: false, up: false };


document.addEventListener('keydown', (e) => {

  if (e.key === 'ArrowRight' || e.key === 'd' || e.key === 'D') keys.right = true;

  if (e.key === 'ArrowLeft' || e.key === 'a' || e.key === 'A') keys.left = true;

  if ((e.key === 'ArrowUp' || e.key === ' ' || e.key === 'w' || e.key === 'W') && !player.jumping) {

    player.vy = -player.jumpStrength;

    player.jumping = true;

  }

});


document.addEventListener('keyup', (e) => {

  if (e.key === 'ArrowRight' || e.key === 'd' || e.key === 'D') keys.right = false;

  if (e.key === 'ArrowLeft' || e.key === 'a' || e.key === 'A') keys.left = false;

});


// Jugador (Dr. Medix)

const player = {

  x: 50,

  y: 300,

  width: 30,

  height: 40,

  speed: 5,

  vy: 0,

  gravity: 0.8,

  jumpStrength: 13,

  jumping: false

};


// Plataformas

const platforms = [

  { x: 0, y: 360, width: 2400, height: 40 }, // Suelo largo

  { x: 200, y: 270, width: 120, height: 20, label: "FESP 1" },

  { x: 400, y: 200, width: 140, height: 20, label: "Determinantes" },

  { x: 650, y: 250, width: 120, height: 20, label: "Incidencia" },

  { x: 900, y: 180, width: 150, height: 20, label: "Prevalencia" },

  { x: 1200, y: 260, width: 130, height: 20, label: "Hanlon" },

  { x: 1500, y: 200, width: 160, height: 20, label: "INEGI / DGIS" }

];


// Monedas FESP (Items)

const coins = [

  { x: 230, y: 230, collected: false, info: "FESP: Monitoreo de la Salud" },

  { x: 450, y: 160, collected: false, info: "Determinante Social Integrado" },

  { x: 680, y: 210, collected: false, info: "Incidencia: Casos Nuevos" },

  { x: 930, y: 140, collected: false, info: "Prevalencia: Casos Totales" },

  { x: 1550, y: 160, collected: false, info: "Fuente de Datos: INEGI/DGIS" }

];


// Enemigos (Reduccionismo Biomédico / Transición Epidemiológica)

const enemies = [

  { x: 350, y: 330, width: 30, height: 30, vx: -2, type: "Biomédico Puro" },

  { x: 800, y: 330, width: 30, height: 30, vx: 2, type: "Sesgo Estadística" },

  { x: 1300, y: 330, width: 35, height: 35, vx: -2.5, type: "E. Crónico-Degenerativa" }

];


// Meta / Centro de Salud

const goal = { x: 1800, y: 260, width: 60, height: 100 };


// Camara

let cameraX = 0;


function update() {

  if (gameOver || gameWon) return;


  // Movimiento Jugador

  if (keys.right) player.x += player.speed;

  if (keys.left && player.x > 0) player.x -= player.speed;


  // Gravedad

  player.vy += player.gravity;

  player.y += player.vy;


  // Colisión con Plataformas

  player.jumping = true;

  platforms.forEach(p => {

    if (

      player.x < p.x + p.width &&

      player.x + player.width > p.x &&

      player.y + player.height > p.y &&

      player.y + player.height - player.vy <= p.y

    ) {

      player.y = p.y - player.height;

      player.vy = 0;

      player.jumping = false;

    }

  });


  // Movimiento Enemigos y Colisión

  enemies.forEach(e => {

    e.x += e.vx;

    if (e.x < 200 || e.x > 1600) e.vx *= -1;


    // Colisión con jugador

    if (

      player.x < e.x + e.width &&

      player.x + player.width > e.x &&

      player.y < e.y + e.height &&

      player.y + player.height > e.y

    ) {

      // Si salta encima, elimina enemigo

      if (player.vy > 0 && player.y + player.height - player.vy <= e.y + 10) {

        e.x = -999;

        player.vy = -8;

      } else {

        gameOver = true;

        message = "¡Caíste ante un " + e.type + "! Revisa tus conceptos.";

      }

    }

  });


  // Recolectar Monedas FESP

  coins.forEach(c => {

    if (!c.collected &&

        player.x < c.x + 20 &&

        player.x + player.width > c.x &&

        player.y < c.y + 20 &&

        player.y + player.height > c.y) {

      c.collected = true;

      score++;

      document.getElementById('score').innerText = `Monedas FESP: ${score} / 5`;

      message = "¡Obtenido!: " + c.info;

    }

  });


  // Llegar a la Meta (Centro de Salud)

  if (

    player.x < goal.x + goal.width &&

    player.x + player.width > goal.x &&

    player.y < goal.y + goal.height &&

    player.y + player.height > goal.y

  ) {

    if (score >= 3) {

      gameWon = true;

      message = "¡FELICIDADES! Has aprobado Salud Pública e integrado la prevención.";

    } else {

      message = "¡Necesitas al menos 3 Monedas FESP para habilitar la Clínica!";

    }

  }


  // Ajustar Cámara

  cameraX = player.x - 200;

  if (cameraX < 0) cameraX = 0;

}


function draw() {

  // Limpiar Pantalla

  ctx.clearRect(0, 0, canvas.width, canvas.height);


  ctx.save();

  ctx.translate(-cameraX, 0);


  // Dibujar Suelo y Plataformas

  platforms.forEach(p => {

    ctx.fillStyle = "#8B4513";

    ctx.fillRect(p.x, p.y, p.width, p.height);

    ctx.fillStyle = "#4CAF50";

    ctx.fillRect(p.x, p.y, p.width, 5); // Pasto superior


    if (p.label) {

      ctx.fillStyle = "#fff";

      ctx.font = "10px Arial";

      ctx.fillText(p.label, p.x + 5, p.y + 15);

    }

  });


  // Dibujar Monedas

  coins.forEach(c => {

    if (!c.collected) {

      ctx.fillStyle = "#FFD700";

      ctx.beginPath();

      ctx.arc(c.x + 10, c.y + 10, 10, 0, Math.PI * 2);

      ctx.fill();

      ctx.fillStyle = "#000";

      ctx.font = "bold 10px Arial";

      ctx.fillText("F", c.x + 7, c.y + 14);

    }

  });


  // Dibujar Enemigos

  enemies.forEach(e => {

    if (e.x > 0) {

      ctx.fillStyle = "#D32F2F";

      ctx.fillRect(e.x, e.y, e.width, e.height);

      // Ojos

      ctx.fillStyle = "#FFF";

      ctx.fillRect(e.x + 5, e.y + 5, 5, 5);

      ctx.fillRect(e.x + 20, e.y + 5, 5, 5);

    }

  });


  // Dibujar Meta (Centro de Salud)

  ctx.fillStyle = "#E0E0E0";

  ctx.fillRect(goal.x, goal.y, goal.width, goal.height);

  ctx.fillStyle = "#D32F2F"; // Cruz Roja

  ctx.fillRect(goal.x + 25, goal.y + 20, 10, 30);

  ctx.fillRect(goal.x + 15, goal.y + 30, 30, 10);

  ctx.fillStyle = "#000";

  ctx.font = "10px Arial";

  ctx.fillText("CLÍNICA DSP", goal.x + 2, goal.y + 70);


  // Dibujar Jugador (Médico/a)

  ctx.fillStyle = "#FFFFFF"; // Bata médica blanca

  ctx.fillRect(player.x, player.y, player.width, player.height);

  ctx.fillStyle = "#FFCC80"; // Cabeza

  ctx.fillRect(player.x + 5, player.y - 10, 20, 10);

  ctx.fillStyle = "#1976D2"; // Estetoscopio

  ctx.fillRect(player.x + 8, player.y + 10, 14, 4);


  ctx.restore();


  // Pantalla / Mensajes de Info

  if (message) {

    ctx.fillStyle = "rgba(0, 0, 0, 0.7)";

    ctx.fillRect(10, 350, 780, 40);

    ctx.fillStyle = "#FFF";

    ctx.font = "14px Arial";

    ctx.fillText(message, 20, 375);

  }


  // Game Over / Victoria

  if (gameOver) {

    ctx.fillStyle = "rgba(0, 0, 0, 0.85)";

    ctx.fillRect(0, 0, canvas.width, canvas.height);

    ctx.fillStyle = "#FF5252";

    ctx.font = "30px Arial";

    ctx.fillText("¡GAME OVER!", 300, 180);

    ctx.font = "16px Arial";

    ctx.fillStyle = "#FFF";

    ctx.fillText("Recarga la página para volver a intentarlo", 260, 220);

  } else if (gameWon) {

    ctx.fillStyle = "rgba(0, 100, 0, 0.85)";

    ctx.fillRect(0, 0, canvas.width, canvas.height);

    ctx.fillStyle = "#FFD700";

    ctx.font = "30px Arial";

    ctx.fillText("¡NIVEL COMPLETADO!", 240, 180);

    ctx.font = "16px Arial";

    ctx.fillStyle = "#FFF";

    ctx.fillText("Has aplicado con éxito las FESP y las medidas epidemiológicas.", 170, 220);

  }

}


function gameLoop() {

  update();

  draw();

  requestAnimationFrame(gameLoop);

}


gameLoop();

</script>
